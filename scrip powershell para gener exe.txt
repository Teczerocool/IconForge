# ==========================================
# Script de Compilación Automática - IconForge
# ==========================================

$ErrorActionPreference = "Stop"

Write-Host "🧹 [1/4] Limpiando carpetas de compilaciones previas..." -ForegroundColor Yellow
Remove-Item -Recurse -Force dist, build -ErrorAction SilentlyContinue

Write-Host "🎨 [2/4] Generando el archivo app_icon.ico desde el código..." -ForegroundColor Yellow
python -c "from PySide6.QtWidgets import QApplication; import sys; app = QApplication(sys.argv); from app_conversor import crear_icono_app; _, pix = crear_icono_app(); pix.save('app_icon.ico', 'ICO')"

if (-not (Test-Path "app_icon.ico")) {
    Write-Host "❌ Error: No se pudo generar app_icon.ico." -ForegroundColor Red
    exit 1
}

Write-Host "📦 [3/4] Compilando ejecutable único con PyInstaller..." -ForegroundColor Yellow
pyinstaller --noconsole `
    --onefile `
    --icon="app_icon.ico" `
    --collect-submodules PIL `
    --collect-submodules PySide6 `
    --collect-submodules PySide6.QtSvg `
    --collect-submodules pillow_heif `
    --collect-submodules pillow_avif `
    --hidden-import=webbrowser `
    --hidden-import=PySide6.QtSvg `
    --hidden-import=pillow_heif `
    --hidden-import=pillow_avif `
    --add-data "icons;icons" `
    app_conversor.py

Write-Host "🚀 [4/4] Renombrando ejecutable a IconForge.exe..." -ForegroundColor Yellow
if (Test-Path "dist/app_conversor.exe") {
    Rename-Item -Path "dist/app_conversor.exe" -NewName "IconForge.exe" -Force
    Write-Host "`n✅ ¡Proceso finalizado con éxito! El ejecutable está en: dist/IconForge.exe" -ForegroundColor Green
} else {
    Write-Host "❌ Error: No se encontró dist/app_conversor.exe al finalizar." -ForegroundColor Red
}
